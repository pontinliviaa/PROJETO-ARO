-- TABELA DE RECEITAS
CREATE TABLE ARO_RECEITA (
    ID_REGISTRO INTEGER PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    DESCRICAO VARCHAR(100) NOT NULL,
    VALOR NUMERIC(12,2) NOT NULL,
    DATA_MOVIMENTO DATE NOT NULL,

    FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO (ID_USUARIO)
);

-- TABELA DE DESPESAS
CREATE TABLE ARO_DESPESA (
    ID_REGISTRO INTEGER PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    DESCRICAO VARCHAR(100) NOT NULL,
    VALOR NUMERIC(12,2) NOT NULL,
    DATA_MOVIMENTO DATE NOT NULL,
    CLASSIFICACAO VARCHAR(8) NOT NULL,

    FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO (ID_USUARIO)
);

-- TABELA DE TAXAS
CREATE TABLE ARO_TAXA (
    ID_TAXA INTEGER PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    PERCENTUAL NUMERIC(5,2) NOT NULL,
    DATA_REFERENCIA DATE DEFAULT CURRENT_DATE NOT NULL UNIQUE,

    FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO (ID_USUARIO)
);

-- TABELA DE SALDOS MENSAIS
CREATE TABLE ARO_SALDO_MENSAL (
    ID_SALDO INTEGER PRIMARY KEY,
    ID_USUARIO INTEGER NOT NULL,
    ANO INTEGER NOT NULL,
    MES INTEGER NOT NULL,
    TOTAL_RECEITAS NUMERIC(18,2) NOT NULL,
    TOTAL_DESPESAS NUMERIC(18,2) NOT NULL,
    SALDO NUMERIC(18,2) NOT NULL,

    UNIQUE (ID_USUARIO, ANO, MES),

    FOREIGN KEY (ID_USUARIO)
        REFERENCES USUARIO (ID_USUARIO)
);
